src/types/logEntry.ts
```typescript
export type Dimension =
  | 'sleep_hours' | 'mood' | 'energy' | 'skin_severity' | 'pain' | 'cravings'
  | 'movement_minutes' | 'bleed_flow' | 'medication_taken' | 'stress'
  | 'digestion' | 'headache' | 'hair_shedding' | 'appetite' | 'notes';

export type CyclePhase = 'menstrual' | 'follicular' | 'ovulatory' | 'luteal' | 'unknown';

export interface LogEntry {
  id: string;            // uuid
  ts: string;            // ISO 8601 datetime
  dayKey: string;        // 'YYYY-MM-DD' — the engine's join key across dimensions
  cyclePhase: CyclePhase;
  dimension: Dimension;
  value: number;
  source: 'tap' | 'voice' | 'seed';
}
```

src/types/finding.ts
```typescript
import type { Dimension } from './logEntry';

export type Finding =
  | { status: 'insufficient_data'; pair: [Dimension, Dimension]; observationCount: number }
  | { status: 'no_pattern'; pair: [Dimension, Dimension]; observationCount: number }
  | {
      status: 'pattern_found';
      pair: [Dimension, Dimension];
      lagDays: number;                      // 1..14
      direction: 'positive' | 'negative';
      strength: number;                      // |r|
      observationCount: number;
      confidenceBand: [number, number];      // percentile bootstrap CI on r
    };
```

src/types/exportPayload.ts
```typescript
import type { Dimension, LogEntry } from './logEntry';
import type { Finding } from './finding';

export interface ExportPayload {
  generatedAt: string;
  rangeStart: string;
  rangeEnd: string;
  cycleHistory: Array<{ start: string; lengthDays: number; type: string }>;
  intervalStats: { medianDays: number; minDays: number; maxDays: number; bleedsPerYear: number };
  symptomTimeline: Array<Pick<LogEntry, 'dayKey' | 'dimension' | 'value'>>;
  medicationLog: Array<{ dayKey: string; name: string; taken: boolean }>;
  findings: Finding[];
  requestedTests: string[];   // fasting insulin, free testosterone, DHEAS, 75g OGTT, AMH
  disclaimer: string;         // app-derived values are wellness trends, not clinical measurements
}
```
